\subsection*{A. Theorems and Proofs}

\stitle{Proposition~\ref{prop-monotone} (Monotone refinement)}.
\begin{proof}
Let $Q$ be a plan with $J = A = \emptyset$ and no slice, and let
$\mu' \in \eval{Q'}{\G}$.
If $Q'$ adds a triple pattern $t$, then by the semantics of join
$\mu' = \mu \cup \nu$ for some compatible $\mu \in \eval{Q}{\G}$ and
$\nu \in \eval{t}{\G}$; $\mu$ is defined exactly on $\beta(Q)$, so the
restriction of $\mu'$ to $\beta(Q)$ is $\mu$.
If $Q'$ adds a filter $e$, then $\eval{Q'}{\G}$ is the set of
mappings of $\eval{Q}{\G}$ that satisfy $e$.
In both cases the restriction of $\mu'$ to $\beta(Q)$ is in
$\eval{Q}{\G}$.
\end{proof}

\subsection*{B. Algorithms and Analysis}

Algorithm~\ref{alg-plan} is the planning loop of
Section~\ref{sec-framework}; Algorithm~\ref{alg-verify} is the
verifier.

\begin{algorithm}[t]
\caption{Verified query planning}
\label{alg-plan}
\KwIn{question $q$, graph $\G$, ontology $O$, policy $\rho$, step budget $B$}
\KwOut{a plan $Q$}
$Q \leftarrow Q_0$; $g \leftarrow \bot$; $h \leftarrow [\,]$\;
\For{$i \leftarrow 1$ \KwTo $B$}{
  $a \leftarrow \rho(q, O, Q, g, h)$\;
  \uIf{$a$ is \kw{goal}}{$g \leftarrow a.g$; $o \leftarrow \textsc{Verify}(Q, g)$\;}
  \uElseIf{$a$ is \kw{lookup} or \kw{inspect}}{$o \leftarrow$ read $\G$\;}
  \uElseIf{$a$ is \kw{finish}}{
    \lIf{$\textsc{Verify}(Q, g)$ has no errors and no open goals}{\Return $Q$}
    $o \leftarrow$ refusal with the open goals\;}
  \Else{
    $Q' \leftarrow a(Q)$; $o \leftarrow \textsc{Verify}(Q', g)$\;
    \If{$o$ has no errors}{
      $o \leftarrow o \cup \{|\eval{Q'}{\G}|$, sample$\}$\;
      \lIf{$Q'$ runs}{$Q \leftarrow Q'$}}}
  append $(a, o)$ to $h$\;
}
\Return the last plan in $h$ that runs\;
\end{algorithm}

\begin{algorithm}[t]
\caption{\textsc{Verify}$(Q, g)$}
\label{alg-verify}
\KwIn{plan $Q$, goal $g$ (possibly $\bot$), ontology $O$}
\KwOut{errors $\mathit{Err}$, typing context $\Gamma_Q$, open goals $\mathit{Open}$}
$\mathit{Err} \leftarrow \emptyset$; $\Gamma_Q \leftarrow \emptyset$\;
\ForEach{triple pattern $(s, p, o)$ of $T$ and $J$}{
  \lIf{$p \notin E$, or a class not in $C$, lies in a namespace $O$ describes}{add to $\mathit{Err}$}
  \lIf{$s$ is a literal, or $o$ is a literal and $\rng(p)$ has a class}{add to $\mathit{Err}$}
  apply \textsc{T-Type}, \textsc{T-Dom}, and \textsc{T-Rng} to extend $\Gamma_Q$\;
}
\lForEach{variable with a type in $\Lit$ and a class in $\Gamma_Q$}{add to $\mathit{Err}$}
check the conditions of Definition~\ref{def-formed} on $\beta(Q)$ and $\beta^+(Q)$, adding violations to $\mathit{Err}$\;
$\mathit{Open} \leftarrow$ the conditions of $Q \models g$ (Definition~\ref{def-goal}) that fail\;
\Return $(\mathit{Err}, \Gamma_Q, \mathit{Open})$\;
\end{algorithm}

\subsection*{C. Evaluation Details}

\stitle{Scorer parity}.
The official Text2SPARQL scorer flattens every value of every variable
into one set and scores it with trec\_eval~\cite{treceval}.
We rescored the predicted result sets of all eight published 2026
entries on CK26 and DB26 and compared every per-question score and
every average with the official files: all 5,024 numbers agree exactly.
The check is a script in the artifact.

\stitle{Quirks of the official metric}.
(1) An ASK answer is scored as one document named \kw{true} with
relevance 1 or 0. A gold \kw{true} therefore scores 1.0 whatever the
system answers, and a gold \kw{false} scores 0.0 for every system.
(2) Every returned value has the same score, so trec\_eval orders
values by their text, and the official nDCG ignores the order the
system returned.
(3) The official $F_{\mathrm{ndcg}}$ average counts the mean nDCG of
the scope once more as an extra item.
Our official column reproduces all three; the strict column scores
Booleans by exact match, uses the returned order for nDCG, and takes a
plain mean.

\stitle{Engine dependence of the CK gold answers}.
Of the seven CK26 gold queries that do not reproduce on Oxigraph, two
break ties under \kw{LIMIT} differently, two use \kw{xsd{:}int()}, which
is not a SPARQL 1.1 function, two format computed numbers differently
(\eg \kw{3} against \kw{3.0}), and one computes a different value for
an arithmetic expression.
On rdflib, 42 of 50 reproduce.
